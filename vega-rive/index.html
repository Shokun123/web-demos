import SwiftUI
import RiveRuntime

struct VegaCompanionView: View {
    @StateObject private var riveVM = RiveViewModel(
        fileName: "vega_companion",
        stateMachineName: "VegaStateMachine",
        artboardName: "VegaCompanion"
    )

    var body: some View {
        riveVM.view()
            .frame(width: 300, height: 300)
    }

    func updateAIState(listening: Bool, speaking: Bool) {
        riveVM.setInput("isListening", value: listening)
        riveVM.setInput("isSpeaking", value: speaking)
    }

    func triggerCelebration() {
        riveVM.triggerInput("celebrate")
    }
}

1. Emotional Legibility Before Words
